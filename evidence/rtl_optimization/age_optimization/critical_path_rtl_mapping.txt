     1	module starvation_aware_arbiter #(
     2	    parameter N = 4,
     3	    parameter AGE_WIDTH = 4,
     4	    parameter AGE_THRESHOLD = 3
     5	)(
     6	    input  wire         clk,
     7	    input  wire         rst,
     8	    input  wire         ready,
     9	    input  wire [N-1:0] req,
    10	    output reg  [N-1:0] grant
    11	);
    12	
    13	    reg [1:0] rr_ptr;
    14	    reg [AGE_WIDTH-1:0] age [0:N-1];
    15	
    16	    reg [N-1:0] aged_req;
    17	    reg [N-1:0] aged_selected;

    21	
    22	    // Select the first asserted request in round-robin order.
    23	    // This implementation is intended for the 4-requester design.
    24	    function [N-1:0] rr_select;
    25	        input [N-1:0] mask;
    26	        input [1:0] ptr;
    27	        begin
    28	            rr_select = {N{1'b0}};
    29	
    30	            case (ptr)
    31	                2'd0: begin
    32	                    if      (mask[0]) rr_select = 4'b0001;
    33	                    else if (mask[1]) rr_select = 4'b0010;
    34	                    else if (mask[2]) rr_select = 4'b0100;
    35	                    else if (mask[3]) rr_select = 4'b1000;
    36	                end
    37	
    38	                2'd1: begin
    39	                    if      (mask[1]) rr_select = 4'b0010;
    40	                    else if (mask[2]) rr_select = 4'b0100;
    41	                    else if (mask[3]) rr_select = 4'b1000;
    42	                    else if (mask[0]) rr_select = 4'b0001;
    43	                end
    44	
    45	                2'd2: begin
    46	                    if      (mask[2]) rr_select = 4'b0100;
    47	                    else if (mask[3]) rr_select = 4'b1000;
    48	                    else if (mask[0]) rr_select = 4'b0001;
    49	                    else if (mask[1]) rr_select = 4'b0010;
    50	                end
    51	
    52	                2'd3: begin
    53	                    if      (mask[3]) rr_select = 4'b1000;
    54	                    else if (mask[0]) rr_select = 4'b0001;
    55	                    else if (mask[1]) rr_select = 4'b0010;
    56	                    else if (mask[2]) rr_select = 4'b0100;
    57	                end
    58	
    59	                default: rr_select = {N{1'b0}};
    60	            endcase
    61	        end
    62	    endfunction
    63	
    64	    // Identify requests that have reached the age threshold.
    65	    always @(*) begin
    66	        for (i = 0; i < N; i = i + 1)
    67	            aged_req[i] = req[i] &&
    68	                          (age[i] >= AGE_THRESHOLD);
    69	    end
    70	
    71	    // Prefer aged requests. If none are aged, use all requests.
    72	    always @(*) begin
    73	        aged_selected = rr_select(aged_req, rr_ptr);
    74	        rr_selected   = rr_select(req, rr_ptr);
    75	
    76	        if (|aged_req)
    77	            grant = ready ? aged_selected : {N{1'b0}};
    78	        else
    79	            grant = ready ? rr_selected : {N{1'b0}};
    80	    end
    81	
    82	    // Update ages and advance the round-robin pointer after a grant.
    83	    always @(posedge clk) begin
    84	        if (rst) begin
    85	            rr_ptr <= 2'd0;
    86	
    87	            for (i = 0; i < N; i = i + 1)
    88	                age[i] <= {AGE_WIDTH{1'b0}};
    89	        end else begin
    90	            for (i = 0; i < N; i = i + 1) begin
    91	                if (!req[i] || (ready && grant[i]))
    92	                    age[i] <= {AGE_WIDTH{1'b0}};
    93	                else if (!grant[i] &&
    94	                         age[i] < AGE_THRESHOLD)
    95	                    age[i] <= age[i] + 1'b1;
    96	            end
    97	
    98	            // Boolean next-state equations for one-hot-or-zero grants.
    99	            rr_ptr[1] <= grant[1] | grant[2] |
   100	                         (rr_ptr[1] & ~(|grant));
   101	            rr_ptr[0] <= grant[0] | grant[2] |
   102	                         (rr_ptr[0] & ~(|grant));
   103	        end
   104	    end
   105	
   106	endmodule
